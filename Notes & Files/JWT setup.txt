private string GenerateToken(User users){
	var securityKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_config["Jwt:Key"]));
            var credentials = new SigningCredentials(securityKey, SecurityAlgorithms.HmacSha256);
            var claims = new[] // optional
            {
                new Claim(ClaimTypes.NameIdentifier,user.Username),
                new Claim(ClaimTypes.Role,user.Role)
		// new Claim(JwtRegisteredClaimNames.Sub, _config["Jwt:Key"]),
		//new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString()),
		//new Claim("UserId", _userData.userId.ToString()),
		//new Claim("", users.username),
		//new Claim("", users.email),
		//new Claim("", users.password)
            };
            var token = new JwtSecurityToken(_config["Jwt:Issuer"],
                _config["Jwt:Audience"],
                claims, // could be null
                expires: DateTime.Now.AddMinutes(15),
                signingCredentials: credentials);


            return new JwtSecurityTokenHandler().WriteToken(token);

}
